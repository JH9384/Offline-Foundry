definition{
  name: admitOwnerCount,
  parameters: [owners : Nat],
  result: (Result Bool AuthorityError),
  body: (if (owners =? 0) then (ctor Result.error [Bool, AuthorityError] (ctor AuthorityError.missingOwner)) else (if (owners =? 1) then (ctor Result.ok [Bool, AuthorityError] true) else (ctor Result.error [Bool, AuthorityError] (ctor AuthorityError.competingOwners))))
}

definition{
  name: admitAuthority,
  parameters: [ownership : Ownership],
  result: (Result Bool AuthorityError),
  body: (match ownership
    | Ownership.product owners => (admitOwnerCount owners)
    | Ownership.external owners boundaryChecked => (match (admitOwnerCount owners)
      | Result.error error => (ctor Result.error [Bool, AuthorityError] error)
      | Result.ok accepted => (if boundaryChecked then (ctor Result.ok [Bool, AuthorityError] true) else (ctor Result.error [Bool, AuthorityError] (ctor AuthorityError.uncheckedBoundary)))))
}
