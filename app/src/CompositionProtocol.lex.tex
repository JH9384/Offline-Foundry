definition{
  name: readNats,
  parameters: [words : (List Nat), count : Nat],
  result: (Option NatsRead),
  body: (match count
    | Nat.zero => (ctor Option.some [NatsRead] (record NatsRead {value := (nil Nat), rest := words}))
    | Nat.succ less => (match words
      | List.nil => (ctor Option.none [NatsRead])
      | List.cons head tail => (match (readNats tail less)
        | Option.none => (ctor Option.none [NatsRead])
        | Option.some parsed => (ctor Option.some [NatsRead] (record NatsRead {value := (cons head (project parsed value)), rest := (project parsed rest)}))))),
  recursive_argument: count
}

definition{
  name: readList,
  parameters: [words : (List Nat)],
  result: (Option NatsRead),
  body: (match words

    | List.cons count tail => (if (count <=? 64) then (readNats tail count) else (ctor Option.none [NatsRead])))
}

definition{
  name: readReference,
  parameters: [words : (List Nat)],
  result: (Option ReferenceRead),
  body: (match words
    | List.nil => (ctor Option.none [ReferenceRead])
    | List.cons v0 t0 => (match t0
      | List.nil => (ctor Option.none [ReferenceRead])
      | List.cons v1 t1 => (ctor Option.some [ReferenceRead] (record ReferenceRead {value := (record FeatureComposition::FeatureRef {key := v0, version := v1}), rest := t1}))))
}

definition{
  name: readInput,
  parameters: [words : (List Nat)],
  result: (Option InputRead),
  body: (match (readReference words)

}

definition{
  name: readFeature,
  parameters: [words : (List Nat)],
  result: (Option FeatureRead),
  body: (match words
    | List.nil => (ctor Option.none [FeatureRead])
    | List.cons key tail0 => (match tail0
      | List.nil => (ctor Option.none [FeatureRead])
      | List.cons version tail1 => (match tail1
        | List.nil => (ctor Option.none [FeatureRead])
        | List.cons cpu tail2 => (match tail2
          | List.nil => (ctor Option.none [FeatureRead])
          | List.cons memory tail3 => (match (readList tail3)
            | Option.none => (ctor Option.none [FeatureRead])
            | Option.some r0 => (match (readList (project r0 rest))
              | Option.none => (ctor Option.none [FeatureRead])
              | Option.some r1 => (match (readList (project r1 rest))
                | Option.none => (ctor Option.none [FeatureRead])
                | Option.some r2 => (match (readList (project r2 rest))
                  | Option.none => (ctor Option.none [FeatureRead])
                  | Option.some r3 => (match (readList (project r3 rest))
                    | Option.none => (ctor Option.none [FeatureRead])
                    | Option.some r4 => (match (readList (project r4 rest))
                      | Option.none => (ctor Option.none [FeatureRead])
                      | Option.some r5 => (match (readList (project r5 rest))
                        | Option.none => (ctor Option.none [FeatureRead])
                        | Option.some r6 => (match (readList (project r6 rest))
                          | Option.none => (ctor Option.none [FeatureRead])
                          | Option.some r7 => (match (readList (project r7 rest))
                            | Option.none => (ctor Option.none [FeatureRead])
                            | Option.some r8 => (match (readList (project r8 rest))
                              | Option.none => (ctor Option.none [FeatureRead])
                              | Option.some r9 => (match (listReferences (project r9 rest))
                                | Option.none => (ctor Option.none [FeatureRead])
                                | Option.some r10 => (match (listInputs (project r10 rest))
                                  | Option.none => (ctor Option.none [FeatureRead])
                                  | Option.some r11 => (ctor Option.some [FeatureRead] (record FeatureRead {value := (record FeatureComposition::FeatureContract {key := key, version := version, vocabulary := (project r0 value), inputs := (project r11 value), outputs := (project r1 value), state := (project r2 value), events := (project r3 value), errors := (project r4 value), effects := (project r5 value), needs := (project r10 value), conflicts := (project r6 value), targets := (project r7 value), accessibility := (project r8 value), quality := (project r9 value), cpu := cpu, memory := memory}), rest := (project r11 rest)}))))))))))))))))))
}

definition{
  name: readPolicy,
  parameters: [words : (List Nat)],

    | List.nil => (ctor Option.none [PolicyRead])

definition{
  name: readWord,

    | Option.none => (ctor Option.none [Nat])
    | Option.some b0 => (match (byteAsNat bytes (offset + 1))
      | Option.none => (ctor Option.none [Nat])
      | Option.some b1 => (match (byteAsNat bytes (offset + 2))
        | Option.none => (ctor Option.none [Nat])
        | Option.some b2 => (match (byteAsNat bytes (offset + 3))
          | Option.none => (ctor Option.none [Nat])
          | Option.some b3 => (ctor Option.some [Nat] (b0 + ((prim multiply b1 256 : Nat) + ((prim multiply b2 65536 : Nat) + (prim multiply b3 16777216 : Nat))))))))),
  axioms: [Quot.sound, propext]
}

definition{
  name: decodeWords,
  parameters: [bytes : Bytes, offset : Nat, fuel : Nat],
  result: (Option (List Nat)),
  body: (match fuel
    | Nat.zero => (if (offset =? (Foundation.Bytes::byteLength bytes)) then (ctor Option.some [(List Nat)] (nil Nat)) else (ctor Option.none [(List Nat)]))
    | Nat.succ less => (if (offset =? (Foundation.Bytes::byteLength bytes)) then (ctor Option.some [(List Nat)] (nil Nat)) else (match (readWord bytes offset)
      | Option.none => (ctor Option.none [(List Nat)])
      | Option.some head => (match (decodeWords bytes (offset + 4) less)
        | Option.none => (ctor Option.none [(List Nat)])
        | Option.some tail => (ctor Option.some [(List Nat)] (cons head tail)))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: dispatchWords,
  parameters: [words : (List Nat)],
  result: Bytes,
  body: (match (readPolicy words)
    | Option.none => 0x6d616c666f726d6564436f6d706f736974696f6e
    | Option.some policy => (match (listFeatures (project policy rest))
      | Option.none => 0x6d616c666f726d6564436f6d706f736974696f6e
      | Option.some features => (match (project features rest)
        | List.cons head tail => 0x6d616c666f726d6564436f6d706f736974696f6e
        | List.nil => (match (FeatureComposition::admitComposition (project policy value) (project features value))
          | Result.error error => (encodeError error)
          | Result.ok admitted => 0x6163636570746564))))
}

definition{
  name: dispatchComposition,
  parameters: [bytes : Bytes],
  result: Bytes,
  body: (if (16384 <? (Foundation.Bytes::byteLength bytes)) then 0x636f6d706f736974696f6e546f6f4c61726765 else (match (decodeWords bytes 0 (Foundation.Bytes::byteLength bytes))
    | Option.none => 0x6d616c666f726d6564436f6d706f736974696f6e
    | Option.some words => (dispatchWords words))),
  axioms: [Quot.sound, propext]
}

definition{
  name: boundary,
  parameters: [],
  result: String,
  body: "Composition wire v1 is compose followed by one ASCII space and little-endian unsigned 32-bit words. Policy is CPU, memory, grants list, targets list, evidence list. Each list begins with its item count. Then comes the feature count and features: key, version, CPU, memory; vocabulary, outputs, state, events, errors, effects, conflicts, targets, accessibility and quality ID lists; dependency key/version pairs; input producer-key/version/schema triples. Maximum payload is 16384 bytes, 32 features, 64 entries per list. Truncation, excess counts and trailing words return malformedComposition; oversized payload returns compositionTooLarge. No semantics are decided by the browser codec. Admission errors and accepted are the generated model result. The wire bounds are not runtime resource budgets."
}
