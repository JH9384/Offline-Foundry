definition{
  name: ruleOpHolds,
  parameters: [op : Nat, bound : Nat, value : Nat],
  result: Bool,
  body: (if (op =? 0) then (value <? bound) else (if (op =? 1) then (value <=? bound) else (if (op =? 2) then (bound <? value) else (if (op =? 3) then (bound <=? value) else (if (op =? 4) then (value =? bound) else (not (value =? bound)))))))
}

definition{
  name: ruleCondsHold,
  parameters: [words : (List Nat), at : Nat, count : Nat, base : Nat],
  result: Bool,
  body: (match count
    | Nat.zero => true
    | Nat.succ less => (if (ruleOpHolds (WorkProtocol::word words (at + 1)) (WorkProtocol::word words (at + 2)) (WorkProtocol::word words (base + (WorkProtocol::word words at)))) then (ruleCondsHold words (at + 3) less base) else false)),
  recursive_argument: count
}

definition{
  name: ruleCondFieldsKnown,
  parameters: [words : (List Nat), at : Nat, count : Nat, fields : Nat],
  result: Bool,
  body: (match count
    | Nat.zero => true
    | Nat.succ less => (if ((WorkProtocol::word words at) <? fields) then (ruleCondFieldsKnown words (at + 3) less fields) else false)),
  recursive_argument: count
}

definition{
  name: ruleSpan,
  parameters: [words : (List Nat), at : Nat],
  result: Nat,
  body: ((at + 2) + (prim multiply 3 (WorkProtocol::word words at) : Nat))
}

definition{
  name: rulesWellFormed,
  parameters: [words : (List Nat), at : Nat, count : Nat, fields : Nat, size : Nat],
  result: Bool,
  body: (match count
    | Nat.zero => true
    | Nat.succ less => (if ((ruleSpan words at) <=? size) then (if (ruleCondFieldsKnown words (at + 2) (WorkProtocol::word words at) fields) then (rulesWellFormed words (ruleSpan words at) less fields size) else false) else false)),
  recursive_argument: count
}

definition{
  name: rulesEnd,
  parameters: [words : (List Nat), at : Nat, count : Nat],
  result: Nat,
  body: (match count | Nat.zero => at | Nat.succ less => (rulesEnd words (ruleSpan words at) less)),
  recursive_argument: count
}

definition{
  name: ruleOutcome,
  parameters: [words : (List Nat), at : Nat, count : Nat, otherwise : Nat, base : Nat],
  result: Nat,
  body: (match count
    | Nat.zero => otherwise
    | Nat.succ less => (if (ruleCondsHold words (at + 2) (WorkProtocol::word words at) base) then (WorkProtocol::word words (at + 1)) else (ruleOutcome words (ruleSpan words at) less otherwise base))),
  recursive_argument: count
}

definition{
  name: ruleDeciding,
  parameters: [words : (List Nat), at : Nat, count : Nat, base : Nat],
  result: Nat,
  body: (match count
    | Nat.zero => 0
    | Nat.succ less => (if (ruleCondsHold words (at + 2) (WorkProtocol::word words at) base) then 0 else ((ruleDeciding words (ruleSpan words at) less base) + 1))),
  recursive_argument: count
}
